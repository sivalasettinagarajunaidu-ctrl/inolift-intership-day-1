import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.metrics import mean_squared_error

# Load PIMA Diabetes Dataset
df = pd.read_csv("diabetes.csv")

# Select Features
features = df.drop("Outcome", axis=1).columns

rmse_values = []

print("===== FEATURE RMSE COMPARISON =====")

# Loop over Features
for feature in features:
    
    rmse = np.sqrt(
        mean_squared_error(
            df["Outcome"],
            df[feature]
        )
    )
    
    rmse_values.append(rmse)
    
    print(f"{feature:25} RMSE = {rmse:.2f}")

# Visualization
colors = ['red','green','blue','orange',
          'purple','cyan','pink','brown']

plt.figure(figsize=(10,5))

plt.bar(
    features,
    rmse_values,
    color=colors
)

plt.axhline(
    y=np.mean(rmse_values),
    color='black',
    linestyle='--',
    label='Average RMSE'
)

plt.title("Diabetes Dataset Feature RMSE Comparison")
plt.xlabel("Features")
plt.ylabel("RMSE")
plt.xticks(rotation=45)
plt.legend()

plt.tight_layout()

plt.savefig("feature_rmse_comparison.png")
plt.show()
